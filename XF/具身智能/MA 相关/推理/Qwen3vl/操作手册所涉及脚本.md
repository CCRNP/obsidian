## Build.sh
``` shell
#!/bin/bash
# ============================================================
# 一键构建 vLLM-Ascend 训推平台自定义镜像
# 镜像含框架 + run.sh，不含模型权重（部署时从 OBS 挂载）
# 模型：Qwen3-VL-32B-Instruct
# ============================================================
set -e

BASE_IMAGE="quay.io/ascend/vllm-ascend:v0.18.0rc1"
IMAGE_NAME="vllm-ascend-modelarts"
IMAGE_TAG="0.18.0rc1"

# SWR 推送目标（镜像名必须小写）
SWR_ADDR="swr.gdrising-global-1.air.gdrising.com.cn"
SWR_ORG="modelarts"
SWR_NAME="qwen3vl-32b-instruct"
SWR_VERSION="v1"
SWR_FULL="${SWR_ADDR}/${SWR_ORG}/${SWR_NAME}:${SWR_VERSION}"

echo "=========================================="
echo "  构建 vLLM-Ascend 训推平台自定义镜像"
echo "  模型：Qwen3-VL-32B-Instruct"
echo "  （run.sh 打入镜像）"
echo "=========================================="
echo "基础镜像: $BASE_IMAGE"
echo "构建镜像: $IMAGE_NAME:$IMAGE_TAG"
echo "SWR目标: $SWR_FULL"
echo ""

if [ ! -f Dockerfile ]; then
    echo "[错误] 请在 Dockerfile 所在目录执行此脚本"
    exit 1
fi

if [ ! -f run.sh ]; then
    echo "[错误] 未找到 run.sh，请确保 run.sh 与 Dockerfile 在同目录"
    exit 1
fi

echo "[步骤 1/3] 构建 Docker 镜像（含 run.sh）..."
docker build -t "${IMAGE_NAME}:${IMAGE_TAG}" .

echo ""
echo "[步骤 2/3] 打标签..."
docker tag "${IMAGE_NAME}:${IMAGE_TAG}" "${SWR_FULL}"

echo ""
echo "[步骤 3/3] 验证镜像..."
docker images | grep -E "vllm-ascend-modelarts|qwen3vl"

```

## run.sh

``` shell
#!/bin/bash
# ============================================================
# vLLM-Ascend 启动脚本
# 打入镜像，位于 /home/ma-user/run.sh（不在 OBS 挂载点，不会被覆盖）
# 模型权重路径：/home/mind/model/（OBS 挂载点）
# ============================================================
set -e

echo "=========================================="
echo "  vLLM-Ascend 启动脚本"
echo "  启动时间: $(date '+%Y-%m-%d %H:%M:%S')"
echo "=========================================="

# ============================================================
# 1. 确定模型权重路径
#    部署在线服务时，平台将 OBS 模型目录挂载到 /home/mind/model/
#    OBS 目录结构：
#      obs://桶/models/qwen3vl/
#      └── weight/          ← 模型权重（含 config.json, safetensors 等）
#
#    挂载后容器内：
#      /home/mind/model/
#      └── weight/
#          ├── config.json
#          ├── *.safetensors
#          └── ...
# ============================================================
MOUNT_DIR="/home/mind/model"

# 优先查找 weight/ 子目录
if [ -d "${MOUNT_DIR}/weight" ] && [ -f "${MOUNT_DIR}/weight/config.json" ]; then
    MODEL_PATH="${MOUNT_DIR}/weight"
    echo "[通过] 模型路径: $MODEL_PATH (weight/ 子目录)"
# 其次检查挂载目录下是否直接有 config.json
elif [ -f "${MOUNT_DIR}/config.json" ]; then
    MODEL_PATH="${MOUNT_DIR}"
    echo "[通过] 模型路径: $MODEL_PATH (挂载根目录)"
else
    echo "[错误] 未找到模型权重"
    echo "[提示] ${MOUNT_DIR}/ 内容:"
    ls -la "${MOUNT_DIR}/" 2>/dev/null || echo "  ${MOUNT_DIR}/ 为空"
    exit 1
fi

# ============================================================
# 2. 检查 NPU 设备
# ============================================================
echo ""
echo "[检查] NPU 设备状态..."
if command -v npu-smi &> /dev/null; then
    npu-smi info 2>&1 | head -20 || echo "[警告] npu-smi 执行失败"
else
    echo "[警告] 未找到 npu-smi 命令，驱动可能未正确挂载"
fi

# ============================================================
# 3. 自动计算张量并行大小
#    平台通过 ASCEND_RT_VISIBLE_DEVICES 注入可见设备
# ============================================================
if [ -n "$ASCEND_RT_VISIBLE_DEVICES" ]; then
    echo "[信息] ASCEND_RT_VISIBLE_DEVICES=$ASCEND_RT_VISIBLE_DEVICES"
    DEVICE_COUNT=$(echo "$ASCEND_RT_VISIBLE_DEVICES" | tr ',' '\n' | wc -l)
    echo "[信息] 可见 NPU 设备数: $DEVICE_COUNT"
    if [ -z "$TP_SIZE" ] || [ "$TP_SIZE" = "1" ]; then
        if [ "$DEVICE_COUNT" -gt 1 ]; then
            export TP_SIZE=$DEVICE_COUNT
            echo "[信息] 自动设置 TP_SIZE=$TP_SIZE"
        fi
    fi
fi

TP_SIZE="${TP_SIZE:-1}"
PORT="${PORT:-8080}"
MAX_MODEL_LEN="${MAX_MODEL_LEN:-32768}"
MAX_NUM_SEQS="${MAX_NUM_SEQS:-128}"
GPU_MEM_UTIL="${GPU_MEM_UTIL:-0.91}"
DTYPE="${DTYPE:-bfloat16}"
SERVED_MODEL_NAME="${SERVED_MODEL_NAME:-qwen3vl-32b}"

# ============================================================
# 4. 打印启动配置
# ============================================================
echo ""
echo "=========================================="
echo "  启动配置"
echo "=========================================="
echo "模型路径:      $MODEL_PATH"
echo "监听端口:      $PORT"
echo "张量并行:      $TP_SIZE"
echo "数据类型:      $DTYPE"
echo "最大序列长度:  $MAX_MODEL_LEN"
echo "最大并发数:    $MAX_NUM_SEQS"
echo "显存利用率:    $GPU_MEM_UTIL"
echo "模型名称:      $SERVED_MODEL_NAME"
echo "=========================================="
echo ""

# ============================================================
# 5. 启动 vLLM 服务
#    端口必须为 8080（手册第5120行，与容器调用接口一致）
#    exec 替换当前进程，确保信号正确传递
# ============================================================
exec vllm serve "$MODEL_PATH" \
    --host 0.0.0.0 \
    --port "$PORT" \
    --dtype "$DTYPE" \
    --tensor-parallel-size "$TP_SIZE" \
    --served-model-name "$SERVED_MODEL_NAME" \
    --max-model-len "$MAX_MODEL_LEN" \
    --max-num-seqs "$MAX_NUM_SEQS" \
    --gpu-memory-utilization "$GPU_MEM_UTIL" \
    --enforce-eager \
    --trust-remote-code

```


## Dockerfile

``` shell
# ============================================================
# 训推平台自定义镜像 - vLLM-Ascend
# 基础镜像：quay.io/ascend/vllm-ascend:v0.18.0rc1
# 适用硬件：910b (Ascend Snt9b)
# 模型来源：从容器镜像中选择（手册第5117行）
# 模型权重：部署在线服务时从 OBS 挂载（手册第5756行）
# run.sh：打入镜像，放在 /home/ma-user/ 下（不在 OBS 挂载点，不会被覆盖）
# ============================================================

FROM quay.io/ascend/vllm-ascend:v0.18.0rc1

LABEL maintainer="AI Platform Team"
LABEL description="vLLM-Ascend for ModelArts HCS 8.6.0"
LABEL version="0.18.0rc1-modelarts-v5"

# ============================================================
# 第一步：创建 ma-user 用户
# 手册第4610行：uid=1000, gid=100
# 手册 MindIE 示例第5964行：useradd -d /home/ma-user -m -u 1000 -g 100 -s /bin/bash ma-user
# 注意：基础镜像 vllm-ascend 已有 gid=100，无需 groupadd
# ============================================================
RUN useradd -d /home/ma-user -m -u 1000 -g 100 -s /bin/bash ma-user

# ============================================================
# 第二步：创建模型挂载目录
# 部署在线服务时，平台将 OBS 模型目录挂载到 /home/mind/model/
# 注意：Docker 挂载会覆盖挂载点原有内容，
#       所以 run.sh 不能放在 /home/mind/model/ 下，放在 /home/ma-user/ 下
# ============================================================
RUN mkdir -p /home/mind/model && \
    chown -R 1000:100 /home/mind && \
    chown -R 1000:100 /home/ma-user

# ============================================================
# 第三步：将 run.sh 打入镜像
# run.sh 位于 /home/ma-user/run.sh（不在 OBS 挂载点，不会被覆盖）
# OBS 挂载到 /home/mind/model/，run.sh 用绝对路径读取模型权重
# ============================================================
COPY run.sh /home/ma-user/run.sh
RUN chmod +x /home/ma-user/run.sh && \
    chown 1000:100 /home/ma-user/run.sh

# ============================================================
# 第四步：NPU 环境变量（镜像内置，平台环境变量可覆盖）
# ============================================================
ENV HCCL_OP_EXPANSION_MODE=AIV
ENV PYTORCH_NPU_ALLOC_CONF=expandable_segments:True
ENV OMP_PROC_BIND=false
ENV OMP_NUM_THREADS=1
ENV TASK_QUEUE_ENABLE=1

# ============================================================
# 第五步：设置工作目录和用户
# 手册 MindIE 示例第5969-5970行
# ============================================================
WORKDIR /home/mind/model/
USER ma-user

# ============================================================
# 第六步：入口命令
# run.sh 在 /home/ma-user/run.sh，不会被 OBS 挂载覆盖
# run.sh 内部用绝对路径 /home/mind/model/ 定位模型权重
# ============================================================
ENTRYPOINT ["/bin/sh", "-c", "bash /home/ma-user/run.sh"]

```
